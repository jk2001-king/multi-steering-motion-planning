cmake_minimum_required(VERSION 3.16)
project(ijcas_nav2_sensor_adapter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Threads REQUIRED)
add_library(ijcas_sensor_nav2_plugins SHARED src/plugins.cpp)
target_compile_options(ijcas_sensor_nav2_plugins PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(ijcas_sensor_nav2_plugins PRIVATE include)
ament_target_dependencies(ijcas_sensor_nav2_plugins nav2_core nav2_costmap_2d rclcpp
  rclcpp_lifecycle pluginlib tf2_geometry_msgs tf2_ros geometry_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ijcas_sensor_nav2_plugins LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
enable_testing()
add_executable(sensor_adapter_contract_test test/contracts.cpp)
target_include_directories(sensor_adapter_contract_test PRIVATE include)
target_link_libraries(sensor_adapter_contract_test Threads::Threads)
add_test(NAME sensor_adapter_contract_test COMMAND sensor_adapter_contract_test)
get_filename_component(IJCAS_WORKSPACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
ament_environment_hooks(env-hooks/ijcas_workspace.dsv.in)
ament_package()
